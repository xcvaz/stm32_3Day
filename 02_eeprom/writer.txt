#include "main.h"
#include <stdio.h>
#include <string.h>

/* Private variables ---------------------------------------------------------*/
UART_HandleTypeDef hlpuart1;
SPI_HandleTypeDef hspi1;

char msg[100];

#define CS_LOW()  HAL_GPIO_WritePin(GPIOA,GPIO_PIN_4,GPIO_PIN_RESET)
#define CS_HIGH() HAL_GPIO_WritePin(GPIOA,GPIO_PIN_4,GPIO_PIN_SET)

/* Function Prototypes */
void SystemClock_Config(void);
static void MX_GPIO_Init(void);
static void MX_LPUART1_UART_Init(void);
static void MX_SPI1_Init(void);

void Flash_WriteEnable(void)
{
	uint8_t cmd=0x06;
	CS_LOW();
	HAL_SPI_Transmit(&hspi1,&cmd,1,100);
	CS_HIGH();
}

uint8_t Flash_Status(void)
{
	uint8_t cmd=0x05;
	uint8_t status;

	CS_LOW();
	HAL_SPI_Transmit(&hspi1,&cmd,1,100);
	HAL_SPI_Receive(&hspi1,&status,1,100);
	CS_HIGH();

	return status;
}

void Flash_WaitBusy(void)
{
	while(Flash_Status()&0x01);
}

void Flash_EraseSector(uint32_t addr)
{
	uint8_t cmd[4];

	Flash_WriteEnable();

	cmd[0]=0x20;
	cmd[1]=addr>>16;
	cmd[2]=addr>>8;
	cmd[3]=addr;

	CS_LOW();
	HAL_SPI_Transmit(&hspi1,cmd,4,100);
	CS_HIGH();

	Flash_WaitBusy();
}

void Flash_Write(uint32_t addr,uint8_t *data,uint16_t len)
{
	uint8_t cmd[4];

	Flash_WriteEnable();

	cmd[0]=0x02;
	cmd[1]=addr>>16;
	cmd[2]=addr>>8;
	cmd[3]=addr;

	CS_LOW();
	HAL_SPI_Transmit(&hspi1,cmd,4,100);
	HAL_SPI_Transmit(&hspi1,data,len,100);
	CS_HIGH();

	Flash_WaitBusy();
}

int main(void)
{
	HAL_Init();
	SystemClock_Config();

	MX_GPIO_Init();
	MX_LPUART1_UART_Init();
	MX_SPI1_Init();

	uint8_t text[]="Hello STM32";

	HAL_Delay(100);

	sprintf(msg,"\r\nFLASH WRITER\r\n");
	HAL_UART_Transmit(&hlpuart1,(uint8_t*)msg,strlen(msg),100);

	sprintf(msg,"Erasing Sector...\r\n");
	HAL_UART_Transmit(&hlpuart1,(uint8_t*)msg,strlen(msg),100);

	Flash_EraseSector(0x000000);

	sprintf(msg,"Writing Data...\r\n");
	HAL_UART_Transmit(&hlpuart1,(uint8_t*)msg,strlen(msg),100);

	Flash_Write(0x000000,text,sizeof(text));

	sprintf(msg,"WRITE SUCCESS\r\n");
	HAL_UART_Transmit(&hlpuart1,(uint8_t*)msg,strlen(msg),100);

	while (1)
	{
	}
}
